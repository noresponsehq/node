# AssessmentAggregate

Body of the 200 answer to POST /v1/assessments/aggregate: one entry per bucket of [from,to), empty buckets included. Counts are of first answers and stay when an assessment or account is deleted or erased; search returns only assessments still retained.

## Example Usage

```typescript
import { AssessmentAggregate } from "@noresponse/node/models";

let value: AssessmentAggregate = {
  schemaVersion: "1",
  from: "2026-10-19T00:00:00.000Z",
  to: "2026-10-26T00:00:00.000Z",
  bucket: "hour",
  groupBy: [
    "asn",
  ],
  metrics: [],
  timeZone: "UTC",
  groupsTruncated: true,
  coverage: {
    asOf: "2026-01-01T00:00:00Z",
    firstAnswersStartedAt: "2026-01-01T00:00:00Z",
    scoreOutcomesStartedAt: "2026-01-01T00:00:00Z",
    retainedFrom: "2026-01-01T00:00:00Z",
    population: "retained_first_answers",
    erasedRecordsIncluded: false,
    overlappingGroups: true,
    lateFlipUnit: "decision_transitions",
    historicalBackfill: false,
  },
  buckets: [
    {
      start: "2026-10-19T00:00:00.000Z",
      groups: [
        {
          country: "GB",
          count: 504589,
        },
      ],
    },
  ],
};
```
