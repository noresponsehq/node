# WebhookEndpoint

One of the workspace's registered webhook endpoints. Signing secrets are not part of it: a secret is shown once, when it is issued (POST /v1/webhooks and POST /v1/webhooks/{endpoint_id}/rotate).

## Example Usage

```typescript
import { WebhookEndpoint } from "@noresponse/node/models";

let value: WebhookEndpoint = {
  endpointId: "whe_01JA2ZD0E1F2G3H4J5K6M7N8P9",
  url: "https://hooks.example.com/noresponse",
  state: "disabled",
  responseMode: "detailed",
  createdAt: "2026-10-22T09:00:00.000Z",
  verifiedAt: "2026-10-22T09:00:05.000Z",
  signingKeys: [
    {
      keyId: "whk_01JA2ZD1M2N3P4Q5R6S7T8V9W0",
      createdAt: "2026-10-22T09:00:00.000Z",
      expiresAt: "2026-10-29T10:00:00.000Z",
    },
  ],
};
```
