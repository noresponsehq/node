# RecordT

One answer from one source: where it came from, what found it and every field it holds. A person's lookup records also say how likely each is the person.

## Example Usage

```typescript
import { RecordT } from "@noresponse/node/models";

let value: RecordT = {
  id: "rec_3f9a1c22d4e5b6a73f9a1c22d4e5b6a7",
  source: {
    connector: "bluesky",
    kind: "platform",
    site: "bsky.app",
    table: "users",
    date: "2019-03",
  },
  matched: [
    {
      by: "handle",
      field: "username",
      search: "srh_01JA2Z8K6Q4V7W9XBCDEFGHJKM",
    },
  ],
  fields: [
    {
      field: "creation_date",
      original: "2019-01-01T00:00:00Z",
      value: "2019-01-01T00:00:00Z",
      detail: "bcrypt",
      distinctive: {
        rating: "unique",
        reason: "in 4 breach rows",
      },
    },
  ],
  observedAt: "2026-01-02T00:00:00Z",
  agreed: [
    "lastip",
  ],
  clashed: [
    "dob",
  ],
};
```
