# Search

One search on a value of one of a person's records, started by a person or by enrichment.

## Example Usage

```typescript
import { Search } from "@noresponse/node/models";

let value: Search = {
  searchId: "srh_01JA2Z8K6Q4V7W9XBCDEFGHJKM",
  recordId: "rec_3f9a1c22d4e5b6a73f9a1c22d4e5b6a7",
  field: "username",
  table: "FORUM_EXAMPLE_COM_2019",
  kind: "username",
  value: "jordanexample42",
  status: "partial",
  sources: 280801,
  answered: 363204,
  found: 358976,
  estimatedSpend: 829179,
  spent: 678805,
  requestedBy: "user_2abc",
  parentSearchId: "srh_01JA2Z8K6Q4V7W9XBCDEFGHJKN",
  createdAt: "2026-01-01T00:00:00Z",
  finishedAt: "2026-01-01T00:00:05Z",
};
```
