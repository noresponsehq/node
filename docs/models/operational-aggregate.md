# OperationalAggregate

One repeatable-read snapshot of retained request diagnostics or webhook attempts. Capture can have gaps and these counts are not a billing ledger. Unattributed keys include older requests and member requests. Queue state is current retained work, not a delivery success denominator.

## Example Usage

```typescript
import { OperationalAggregate } from "@noresponse/node/models";

let value: OperationalAggregate = {
  schemaVersion: "1",
  source: "webhook_attempts",
  from: "2026-01-01T00:00:00Z",
  to: "2026-01-01T00:00:00Z",
  asOf: "2026-01-01T00:00:00Z",
  retainedFrom: "2026-01-01T00:00:00Z",
  retentionDays: 387166,
  bucket: "hour",
  capture: "best_effort",
  population: "retained_attempts",
  totals: {
    attempts: 636679,
    succeeded: 119463,
    failed: 387056,
    clientErrors: 453934,
    serverErrors: 849488,
    replays: 343356,
    incompleteWrites: 157628,
    unattributedKeys: 449916,
  },
  buckets: [],
  queue: {
    pending: 161093,
    leased: 822407,
    deadRetained: 378294,
    oldestPendingAt: "2026-01-01T00:00:00Z",
  },
};
```
