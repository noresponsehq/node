# AuthContext

Normalised claims from a session your backend has already verified with WorkOS, Clerk, Auth0 or any OIDC provider. Pass the output of the SDK's auth adapters, not a raw token.

## Example Usage

```typescript
import { AuthContext } from "@noresponse/node/models";

let value: AuthContext = {
  provider: "clerk",
  issuer: "https://example-app.clerk.accounts.dev",
  audience: "https://app.example.com",
  audiences: [
    "https://app.example.com",
  ],
  subject: "user_2EXAMPLE",
  sessionId: "sess_2EXAMPLE",
  organizationId: "org_2EXAMPLE",
  tenantId: "tenant_2EXAMPLE",
  tokenId: "2EXAMPLEJTI",
  issuedAt: "2026-10-14T08:59:40.000Z",
  expiresAt: "2026-10-14T09:00:40.000Z",
  sessionCreatedAt: "2026-10-14T08:57:00.000Z",
  authTime: "2026-10-14T08:55:00.000Z",
  accountCreatedAt: "2026-10-14T08:56:00.000Z",
  verified: true,
  mfa: {
    methods: [
      "totp",
    ],
  },
  roles: [
    "admin",
  ],
  scopes: [
    "org:sys_profile:manage",
  ],
};
```
