# OtherEvent

An attendee of another event in the same environment who shares an email or phone with this one, as that event recorded them.

## Example Usage

```typescript
import { OtherEvent } from "@noresponse/node/models";

let value: OtherEvent = {
  eventId: "evn_01J9Y3C2M4N6P8Q0RSTVWXYZAB",
  eventName: "Spring gala",
  eventStartsAt: "2025-09-01T09:00:00Z",
  attendeeId: "att_01J9Y3C2M4N6P8Q0RSTVWXYZAB",
  matchedOn: [],
  matchState: "confirmed",
  highestBand: "low",
  decision: "escalated",
  decidedBy: "user_2abc",
  decidedAt: "2025-09-01T12:00:00Z",
};
```
